راهنمای فعال‌سازی خروجی JSON در Google Apps Script (کد doGet) ▼

مطمئن شوید شیت شما دارای تب Journal و ستون‌های Date, Symbol, Status, ExitPrice, ProfitUSD, Lots, Risk است. سپس تابع زیر را در پروژه Apps Script خود قرار دهید و یک New Deployment با دسترسی Anyone بسازید: 

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Journal");
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var trades = [];
  for (var i = 1; i < data.length; i++) {
    var row = {};
    for (var j = 0; j < headers.length; j++) {
      row[headers[j]] = data[i][j];
    }
    trades.push(row);
  }
  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    trades: trades
  })).setMimeType(ContentService.MimeType.JSON);
}
